---
title: Execute a large order
description: Accept a split plan and run it leg by leg, one signature at a time.
---

## Prerequisites

- The [TypeScript project](/guides/typescript-setup) which provides `src/evm.ts`.
- Read [Large orders](/concepts/large-orders).

## Script

```typescript src/large-order.ts
import { bridg, signEvmStep, account } from './evm';
import type { BuildResponse } from '@kernlog/bridg-sdk';

async function signAndSubmit(build: BuildResponse) {
  let mainHash: `0x${string}` | undefined;
  for (const step of build.steps) {
    const hash = await signEvmStep(step);
    if (step.step === 'main') mainHash = hash;
  }
  await bridg.submitTransfer(build.transferId, { txHash: mainHash!, step: 'main' });
  return bridg.waitForTransfer(build.transferId);
}

async function main() {
  const quote = await bridg.getQuote({
    fromChain: 'base',
    toChain: 'arbitrum',
    fromToken: 'USDC',
    toToken: 'USDC',
    amountAtomic: '250000000000', // 250,000 USDC
    sender: account.address,
    recipient: account.address,
  });

  if (!quote.splitPlan) {
    console.log(`single route (${quote.planReason ?? 'not evaluated'})`);
    return signAndSubmit(await bridg.buildTransfer({ decisionId: quote.decisionId }));
  }
  console.log(`split beats the best single venue by ${quote.splitPlan.improvementBps} bps, at most ${quote.splitPlan.maxSignatures} signatures`);

  await bridg.acceptPlan({ decisionId: quote.decisionId, plan: 'split' });
  let group = await bridg.startSplit({ decisionId: quote.decisionId });

  while (group.status === 'RUNNING' && group.next) {
    const leg = await signAndSubmit(group.next);
    console.log(`leg ${group.legs.length} ${leg.venue} ${leg.status}`);
    // The remainder is re-quoted against live prices; the next venue may differ from the plan.
    group = await bridg.nextSplitLeg({ groupId: group.groupId });
  }
  console.log(group.status, `saved ${group.savedVsBaselineBps} bps vs ${group.baseline.venue}`);
}

main().catch((e) => { console.error(e); process.exit(1); });
```

## Notes

- Only a firm, exact-input quote above the asset's floor is probed for a split.
- `nextSplitLeg` while a leg is still in flight is `409 leg_in_flight`.
- `savedVsBaselineBps` can be negative; the order reports what happened, not what was planned.
